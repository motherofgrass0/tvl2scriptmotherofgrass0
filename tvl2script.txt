local function is_hub_key(k)
	if type(k) ~= "string" then return false end
	if k == "__ml_v2" or k == "__ml_v2_prev" or k == "__coverhub_hub" or k == "__coverhub_osp"
		or k == "__coverhub_cos" or k == "__coverhub_recolor" or k == "__coverhub_unload"
		or k == "__cvh_shared" or k == "__MLAim" or k == "_tvl" or k == "COVERHUB_KEY" or k == "COVERHUB_HWID" then
		return true
	end
	local lk = k:lower()
	return lk:find("coverhub", 1, true) ~= nil or lk:find("tvl", 1, true) ~= nil
		or lk:find("mlhub", 1, true) ~= nil or lk:find("__ml", 1, true) ~= nil
end

local function filter_reg(list)
	local out = {}
	for _, item in ipairs(list) do
		local keep = true
		if type(item) == "function" then
			local ok, info_src = pcall(debug.info, item, "s")
			if ok and type(info_src) == "string" and (info_src:find("ml_v2", 1, true) or info_src:find("coverhub", 1, true) or info_src:find("MCP", 1, true) or info_src == "") then
				keep = false
			end
		elseif type(item) == "table" then
			for k in pairs(item) do
				if is_hub_key(k) then
					keep = false
					break
				end
			end
		end
		if keep then out[#out + 1] = item end
	end
	return out
end

local function build_sandbox(base_env)
	local empty_fn = function() end
	local fake_g = setmetatable({}, {
		__index = function(_, k)
			if is_hub_key(k) then return nil end
			return rawget(_G, k)
		end,
		__newindex = empty_fn,
		__iter = function()
			return function(_, prev)
				local k, v = next(_G, prev)
				while k ~= nil do
					if not is_hub_key(k) then return k, v end
					k, v = next(_G, k)
				end
			end
		end,
	})
	local fake_dbg = setmetatable({}, {
		__index = function(_, k)
			local real = rawget(debug, k)
			if k == "info" then
				return function(a, b, c)
					local ok, res = pcall(real, a, b, c)
					if not ok then return nil end
					if type(res) == "string" and (res:find("ml_v2", 1, true) or res:find("coverhub", 1, true) or res:find("MCP", 1, true)) then
						return "ReplicatedStorage.Vendor.Janitor"
					end
					return res
				end
			end
			if k == "traceback" then return function() return "stack traceback:\n" end end
			if k == "getupvalues" or k == "getupvalue" or k == "getconstants" or k == "getconstant" or k == "getprotos" or k == "getproto" then
				return function() return nil end
			end
			return real
		end,
	})
	return setmetatable({}, {
		__index = function(_, k)
			if k == "_G" or k == "shared" then return fake_g end
			if k == "getfenv" or k == "getgenv" or k == "getrenv" then return function() return fake_g end end
			if k == "getgc" or k == "filtergc" then return function() return {} end end
			if k == "getreg" then
				return function()
					local ok, raw = pcall(getreg)
					return (ok and type(raw) == "table") and filter_reg(raw) or {}
				end
			end
			if k == "debug" then return fake_dbg end
			if k == "hookfunction" or k == "hookmetamethod" or k == "replaceclosure" or k == "newcclosure" then return empty_fn end
			if k == "getscripts" or k == "getloadedmodules" or k == "getsenv" or k == "getrunningscripts" then
				return function() return {} end
			end
			if is_hub_key(k) then return nil end
			return base_env[k]
		end,
		__newindex = function(_, k, v) base_env[k] = v end,
	})
end

local cached_tracks = setmetatable({}, { __mode = "k" })
local function preload_anims(ch)
    if not ch then return end
    task.spawn(function()
        local hum = ch:WaitForChild("Humanoid", 3)
        local anm = hum and hum:WaitForChild("Animator", 3)
        if not anm then return end
        cached_tracks[anm] = cached_tracks[anm] or {}
        for id, obj in pairs(ANIM_OBJS) do
            if not cached_tracks[anm][id] then
                local ok, tr = pcall(anm.LoadAnimation, anm, obj)
                if ok and tr then
                    tr.Priority = Enum.AnimationPriority.Action
                    cached_tracks[anm][id] = tr
                end
            end
        end
    end)
end

local function playSpecificAnimation(animId)
    if unloaded then return end
    task.spawn(function()
        if unloaded then return end
        local character = lp.Character or lp.CharacterAdded:Wait()
        local humanoid = character:FindFirstChildOfClass("Humanoid") or character:WaitForChild("Humanoid", 2)
        local animator = humanoid and (humanoid:FindFirstChildOfClass("Animator") or humanoid:WaitForChild("Animator", 2))
        if not animator then return end

        local animation = ANIM_OBJS[animId] or Instance.new("Animation")
        animation.AnimationId = animId

        local ok, track = pcall(function()
            return (cached_tracks[animator] and cached_tracks[animator][animId]) or animator:LoadAnimation(animation)

local wrap_cb
wrap_cb = function(ts, orig)
    local kind = ts.TargetType
    local wrapper = function(info, victim)
        local cur = get_aim_target()
        if not cur then
            return orig(info, victim)
        end
        if is_tool_ts(ts, orig, info) then return orig(info, victim) end
        local a = anchor(cur)
        local name = (type(info) == "table" and rawget(info, "_name")) or (ability_hdl and type(rawget(ability_hdl, "activeAbility")) == "table" and rawget(rawget(ability_hdl, "activeAbility"), "_name"))
        if name and (AIM_EXCLUDED[name] or (tool_data and tool_data[name])) then return orig(info, victim) end
        if not a then return orig(info, victim) end
        if kind == AREA_CHECK or kind == CYLINDER_CAST then
            return { [cur] = true }
        elseif kind == CAST_RESULT then
            return a.Position
        elseif victim ~= nil then
            if victim == cur then return cur end
            return nil
        else
            return cur, nil
        end
    end
    return newcclosure and newcclosure(wrapper) or wrapper
end

local function patch_ts(ts)
    if patched[ts] or type(ts) ~= "table" or not ts.Callback then return end
    if is_tool_ts(ts, ts.Callback, nil) then return end
    patched[ts] = ts.Callback
    ts.Callback = wrap_cb(ts, ts.Callback)
end

    while true do

local function is_pos_inside_outer_circle(outer_circle, pos)
    if not (outer_circle and outer_circle:IsA("BasePart") and pos) then return false end
    local tr = tonumber(outer_circle.Transparency) or 0
    if tr ~= 0 then return false end

    local rel = outer_circle.CFrame:PointToObjectSpace(pos)
    local sz = outer_circle.Size
    local rad_x = sz.X * 0.5
    local rad_z = sz.Z * 0.5
    local rad = math.max(rad_x, rad_z)
    if rad <= 0.1 then return false end

    local half_h = math.max(sz.Y * 0.5, 12)
    if math.abs(rel.Y) > half_h then return false end

    local norm_dist = math.sqrt((rel.X / math.max(rad_x, 0.1)) ^ 2 + (rel.Z / math.max(rad_z, 0.1)) ^ 2)
    return norm_dist <= 1.02
end

    return is_pos_inside_outer_circle(outer, pos)

local motus_cd_until = 0
local motus_cd_triggered_for_cycle = false

local function get_cd_label()
    local slot = get_toolbar_slot()
    if not slot then return nil end

    local cd_lbl = slot_cd_map[slot]
    if not (cd_lbl and cd_lbl.Parent and math.floor(cd_lbl.TextSize + 0.5) == 42) then
        cd_lbl = nil
        for _, v in ipairs(slot:GetChildren()) do
            if v:IsA("TextLabel") and math.floor(v.TextSize + 0.5) == 42 then
                cd_lbl = v
                break
            end
        end
        if not cd_lbl then
            for _, v in ipairs(slot:GetDescendants()) do
                if v:IsA("TextLabel") and math.floor(v.TextSize + 0.5) == 42 then
                    cd_lbl = v
                    break
                end
            end
        end
        slot_cd_map[slot] = cd_lbl
    end

    if cd_lbl and cd_lbl ~= watched_cd_lbl then
        if watched_cd_conn then
            watched_cd_conn:Disconnect()
            watched_cd_conn = nil
        end
        watched_cd_lbl = cd_lbl
        watched_cd_conn = add_conn(cd_lbl:GetPropertyChangedSignal("Visible"):Connect(function()
            if unloaded then return end
            if cd_lbl.Visible then
                if is_motus_equipped() and not motus_cd_triggered_for_cycle then
                    motus_cd_triggered_for_cycle = true
                    motus_cd_until = os.clock() + 15
                    trigger_motus_cooldown_backup()
                end
            else
                motus_cd_triggered_for_cycle = false
                local h = ability_hdl and rawget(ability_hdl, "activeAbility")
                local n = type(h) == "table" and rawget(h, "_name") or nil
                if n then clear_cd_debounce(n) end
                osp_acc = 1
                if trigger_instant_spam then trigger_instant_spam() end
            end
        end))
    end
    return cd_lbl
end

local function is_ability_on_cooldown(ab_name)
    if is_motus_equipped(ab_name) and os.clock() < motus_cd_until then
        return true
    end
    local cd_lbl = get_cd_label()
    return cd_lbl ~= nil and cd_lbl.Visible == true
end

local function is_self_blocked()
    local ch = lp.Character
    if not ch then return true end
    bind_self_watchers(ch)
    local hum = ch:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return true end
    if ch:GetAttribute("ragdoll") == true then return true end

    local anm = hum:FindFirstChildOfClass("Animator")
    if anm and type(anm.GetPlayingAnimationTracks) == "function" then
        local ok, tracks = pcall(anm.GetPlayingAnimationTracks, anm)
        if ok and type(tracks) == "table" then
            for _, tr in ipairs(tracks) do
                if tr.IsPlaying and tr.Animation then
                    local aid = tostring(tr.Animation.AnimationId or "")
                    if is_walkspeed_gate_anim_id(aid) or is_end_only_anim_id(aid) then
                        if not watched_anim_tracks[tr] then
                            watched_anim_tracks[tr] = true
                            tr.Stopped:Once(function()
                                watched_anim_tracks[tr] = nil
                                if unloaded then return end
                                osp_acc = 1
                                if trigger_instant_spam then trigger_instant_spam() end
                            end)
                        end
                        return true
                    end
                end
            end
        end
    end
    return false
end

local function is_target_playing_blocked_anim(mdl, h, ab_name)
    if not mdl or is_aoe_ability(h, ab_name) then return false end
    local hum = mdl:FindFirstChildOfClass("Humanoid")
    if not hum then return false end

    local anm = hum:FindFirstChildOfClass("Animator") or mdl:FindFirstChildOfClass("AnimationController")
    if anm and type(anm.GetPlayingAnimationTracks) == "function" then
        local ok, tracks = pcall(anm.GetPlayingAnimationTracks, anm)
        if ok and type(tracks) == "table" then
            for _, tr in ipairs(tracks) do
                if tr.IsPlaying and tr.Animation then
                    local aid = tostring(tr.Animation.AnimationId or "")
                    if is_end_only_anim_id(aid) then
                        end_only_triggered_entities[mdl] = true
                        if not watched_anim_tracks[tr] then
                            watched_anim_tracks[tr] = true
                            tr.Stopped:Once(function()
                                watched_anim_tracks[tr] = nil
                                end_only_triggered_entities[mdl] = nil
                                if unloaded then return end
                                osp_acc = 1
                                if trigger_instant_spam then trigger_instant_spam() end
                            end)
                        end
                    elseif is_walkspeed_gate_anim_id(aid) then
                        stun_triggered_entities[mdl] = true
                    end
                end
            end
        end
    end

    if end_only_triggered_entities[mdl] then
        return true
    end

    if not stun_triggered_entities[mdl] then
        return false
    end

    if not watched_ws_hums[hum] then
        watched_ws_hums[hum] = true
        add_conn(hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
            if unloaded then return end
            if hum.WalkSpeed ~= 0 and stun_triggered_entities[mdl] then
                stun_triggered_entities[mdl] = nil
                osp_acc = 1
                if trigger_instant_spam then trigger_instant_spam() end
            end
        end))
    end

    if hum.WalkSpeed == 0 then
        return true
    else
        stun_triggered_entities[mdl] = nil
        return false
    end
end

local function is_target_blocked(t_name, h, ab_name, is_proj_cast)
    if not t_name or t_name == "None" then return true end
    if is_spawn_protected(t_name) then return true end

    local mdl = (watched_rag_name == t_name and watched_rag_ent and watched_rag_ent.Parent) and watched_rag_ent or nil
    if not mdl then
        local ent = ws:FindFirstChild("Entities")
        if not ent then return false end
        mdl = ent:FindFirstChild(t_name)
        if not mdl then
            local p = plrs:FindFirstChild(t_name)
            if p and p.Character and p.Character:IsDescendantOf(ent) then mdl = p.Character end
        end
        if not mdl then return false end
        if mdl ~= watched_rag_ent then
            if watched_rag_conn then watched_rag_conn:Disconnect(); watched_rag_conn = nil end
            watched_rag_ent = mdl
            watched_rag_name = t_name
            watched_rag_conn = add_conn(mdl:GetAttributeChangedSignal("ragdoll"):Connect(function()
                if unloaded then return end
                local cur_h = ability_hdl and rawget(ability_hdl, "activeAbility")
                local nm = type(cur_h) == "table" and rawget(cur_h, "_name") or nil
                local rag = mdl:GetAttribute("ragdoll") == true
                local unblock = false
                if is_ictus_equipped(nm) or is_motus_equipped(nm) then
                    unblock = true
                elseif proj_on then
                    unblock = (not is_sol_blind(nm)) or (not rag)
                elseif is_heel_stomp(nm) then
                    unblock = rag
                else
                    unblock = not rag
                end
                if unblock then
                    osp_acc = 1
                    if trigger_instant_spam then trigger_instant_spam() end
                end
            end))
        end
    end

    local hrp = mdl:FindFirstChild("HumanoidRootPart") or mdl.PrimaryPart
    if is_target_in_incendias_outer_circle(t_name, mdl, hrp) then
        return true
    end

    local is_rag = mdl:GetAttribute("ragdoll") == true
    local proj_mode = is_proj_cast or proj_on

    if is_ictus_equipped(ab_name) or is_motus_equipped(ab_name) then
    elseif proj_mode then
        if is_sol_blind(ab_name) and is_rag then
            return true
        end
    elseif is_heel_stomp(ab_name) then
        if not is_rag then return true end
    else
        if is_rag then return true end
    end

    if is_target_playing_blocked_anim(mdl, h, ab_name) then
        return true
    end
    return false
end

local function is_in_animation_range(ab_name, eq_id, my_real_pos, target_hrp)
    if not (my_real_pos and target_hrp) then return false end
    local d = ab_name and ability_data and ability_data[ab_name]
    local base_r = (d and (tonumber(d.range) or tonumber(d.Range) or tonumber(d.maxRange)))
        or (eq_id == fire_circle and 25 or 50)
    local max_r = base_r + 6

    local diff = target_hrp.Position - my_real_pos
    local horiz_dist = Vector3.new(diff.X, 0, diff.Z).Magnitude
    local vert_dist = math.abs(diff.Y)

    return (diff.Magnitude <= max_r) or (horiz_dist <= max_r and vert_dist <= (max_r + 35))
end

local function is_within_strict_spam_range(ab_name, eq_id, target_hrp)
    if not is_ictus_equipped(ab_name) and not is_fire_circle_equipped(ab_name) then
        return true
    end
    local lch = lp.Character
    local my_root = lch and (lch:FindFirstChild("HumanoidRootPart") or lch.PrimaryPart)
    if not (my_root and target_hrp) then return false end

    local d = ab_name and ability_data and ability_data[ab_name]
    local r = (d and (tonumber(d.range) or tonumber(d.Range) or tonumber(d.maxRange)))
        or (is_fire_circle_equipped(ab_name) and 22 or 28)
    return (target_hrp.Position - my_root.Position).Magnitude <= (r + 1.5)
end

local last_anim_t = {}
local function play_ability_animation(ab_name, eq_id, my_real_pos, target_hrp)
    if unloaded or is_ability_on_cooldown(ab_name) then return end

    local aid = ABILITY_ANIMS[norm_key(ab_name)]
    if not aid and ability_data and ab_name and ability_data[ab_name] then
        local d = ability_data[ab_name]
        aid = ABILITY_ANIMS[norm_key(d.displayName or d.DisplayName or d.name or d.Name)]
    end
    if not aid and eq_id == fire_circle then aid = ABILITY_ANIMS["firecircle"] end
    if not aid then return end

    if not is_in_animation_range(ab_name, eq_id, my_real_pos, target_hrp) then
        return
    end

    local h = ability_hdl and rawget(ability_hdl, "activeAbility")
    if cur_tgt_ch and cur_tgt_ch.Parent and is_target_playing_blocked_anim(cur_tgt_ch, h, ab_name) then
        return
    end

    local ch = lp.Character
    local hum = ch and ch:FindFirstChildOfClass("Humanoid")
    local anm = hum and hum.Health > 0 and hum:FindFirstChildOfClass("Animator")
    if not anm then return end

    cached_tracks[anm] = cached_tracks[anm] or {}
    local tr = cached_tracks[anm][aid]
    if not tr then
        local obj = ANIM_OBJS[aid]
        if obj then
            local ok, loaded = pcall(anm.LoadAnimation, anm, obj)
            if ok and loaded then
                loaded.Priority = Enum.AnimationPriority.Action
                cached_tracks[anm][aid] = loaded
                tr = loaded
            end
        end
    end
    if tr and not is_ability_on_cooldown(ab_name) and not unloaded then
        local now = os.clock()
        if not last_anim_t[aid] or (now - last_anim_t[aid]) >= 0.08 or not tr.IsPlaying then
            last_anim_t[aid] = now
            pcall(function() tr:Play(0.05, 1, 1) end)
        end
    end
end

local function get_char(p)
    if not p or p == lp then return nil, nil end
    if is_spawn_protected(p.Name) then return nil, nil end
    local ent = ws:FindFirstChild("Entities")
    if not ent then return nil, nil end
    local ch = ent:FindFirstChild(p.Name) or (p.Character and p.Character:IsDescendantOf(ent) and p.Character)
    if not ch then return nil, nil end
    if ch:GetAttribute("IsTeleported") == true or ch:GetAttribute("isTeleported") == true then return nil, nil end
    if ch:FindFirstChildOfClass("ForceField") then return nil, nil end
    for _, a in ipairs({ "GodMode","ProtectionMode","InPendantTrap","PendantTrap","InAstralProjection" }) do
        if ch:GetAttribute(a) == true or p:GetAttribute(a) == true then return nil, nil end
    end
    local slot = p:GetAttribute("CharacterName") or ch:GetAttribute("CharacterName") or ch:GetAttribute("SpecieType")
    if slot == "" then return nil, nil end
    local hrp = ch:FindFirstChild("HumanoidRootPart") or ch.PrimaryPart
    local hum = ch:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return nil, nil end
    return ch, hrp
end

    local left_str = string.match(txt, "^%s*([%d,%.]+)%s*/")
    if not left_str then return nil end
    return tonumber((left_str:gsub(",", "")))
end

local function on_magic_text_changed(lbl)
    if unloaded then return end
    local cur_val = parse_left_magic(lbl.Text)
    if not cur_val then return end
    if last_magic_val ~= nil then
        local diff = last_magic_val - cur_val
        if (diff == 30 or (diff >= 28 and diff <= 32)) and is_ictus_equipped() and is_any_target_in_ability_range("Ictus") then
            playIctusAnimation()
        end
        if (diff == 60 or (diff >= 58 and diff <= 62)) and is_motus_equipped() and is_any_target_in_ability_range("Motus") then
            motus_cd_triggered_for_cycle = true
            motus_cd_until = os.clock() + 15
            trigger_motus_cooldown_backup()
            trigger_motus_cooldown_and_anim()
        end
    end
    last_magic_val = cur_val
end

local function get_ability()
    init_caches()
    local best = get_toolbar_slot()
    if not best then cur_eq_id = nil; return false, false, nil, nil, nil end
    local icn
    for _, v in ipairs(best:GetChildren()) do
        if v:IsA("ImageLabel") and v.Name == "ImageLabel" and v.Visible then icn = v; break end
    end
    if not icn then
        for _, v in ipairs(best:GetChildren()) do
            if v:IsA("ImageLabel") and v.Name ~= "Burst" and v.Visible then icn = v; break end
        end
    end
    if not icn or icn.Image == "" then cur_eq_id = nil; return false, false, nil, "", icn and icn.ImageColor3 end
    local id = string.match(icn.Image, "%d+")
    cur_eq_id = id

    local h = ability_hdl and rawget(ability_hdl, "activeAbility")
    local h_name = type(h) == "table" and rawget(h, "_name") or nil
    local ts = type(h) == "table" and rawget(h, "_targetSystem") or nil
    local wl = is_heel_stomp(h_name) or is_aoe_ability(h, h_name) or is_ictus_equipped(h_name) or is_motus_equipped(h_name) or is_no_highlight_aoe_equipped(h_name)
    if not wl and type(h_name) == "string" then
        local nk = norm_key(h_name)
        if ABILITY_ANIMS[nk] then wl = true
        elseif ability_data and ability_data[h_name] then
            local dn = norm_key(ability_data[h_name].displayName or ability_data[h_name].name or "")
            if ABILITY_ANIMS[dn] then wl = true end
        end
    end
    if not id or (bad_ids[id] and not wl) then return false, false, id, icn.Image, icn.ImageColor3 end
    if id == fire_circle and hover_self() then return false, false, id, icn.Image, icn.ImageColor3 end
    local is_proj = proj_ids[id] == true or (ts and target_type and (ts.TargetType == CAST_RESULT or (h_name and force_normal[h_name]))) or false
    return true, is_proj, id, icn.Image, icn.ImageColor3
end

local los_rp = RaycastParams.new()
los_rp.FilterType = Enum.RaycastFilterType.Exclude
los_rp.IgnoreWater = true

local function path_clear(o, t, ign)
    local flt = table.clone(ign)
    local cur = o
    for _ = 1, 8 do
        local dir = t - cur
        if dir.Magnitude <= 0.5 then return true end
        los_rp.FilterDescendantsInstances = flt
        local res = ws:Raycast(cur, dir, los_rp)
        if not res then return true end
        local p = res.Instance
        if p and not p.CanCollide and p.Transparency >= 0.95 then
            table.insert(flt, p)
            cur = res.Position + dir.Unit * 0.05
        else
            return false
        end
    end
    return false
end

local function has_los(ch, hrp)
    if not resp_barriers then return true end
    cam = ws.CurrentCamera
    if not cam then return false end
    local lch = lp.Character
    local lhrp = lch and (lch:FindFirstChild("HumanoidRootPart") or lch.PrimaryPart)
    local ent = ws:FindFirstChild("Entities")
    local ign = { lch, ch, ent }
    if not path_clear(cam.CFrame.Position, hrp.Position, ign) then return false end
    if lhrp and not path_clear(lhrp.Position, hrp.Position, ign) then return false end
    return true
end

local bldg_rp = RaycastParams.new()
bldg_rp.FilterType = Enum.RaycastFilterType.Exclude
bldg_rp.IgnoreWater = true
local bldg_ovp = OverlapParams.new()
bldg_ovp.FilterType = Enum.RaycastFilterType.Exclude
local DIRS = {
    Vector3.new(1,0,0), Vector3.new(-1,0,0), Vector3.new(0,0,1), Vector3.new(0,0,-1),
    Vector3.new(0.707,0,0.707), Vector3.new(-0.707,0,0.707), Vector3.new(0.707,0,-0.707), Vector3.new(-0.707,0,-0.707)
}

local function ray_solid(o, d, ign)
    local flt = table.clone(ign)
    local cur, rem = o, d.Magnitude
    if rem <= 0.05 then return nil end
    local u = d.Unit
    for _ = 1, 8 do
        bldg_rp.FilterDescendantsInstances = flt
        local r = ws:Raycast(cur, u * rem, bldg_rp)
        if not r then return nil end
        if r.Instance and r.Instance.CanCollide then return r
        elseif r.Instance then
            table.insert(flt, r.Instance)
            rem = rem - (r.Position - cur).Magnitude - 0.05
            cur = r.Position + u * 0.05
            if rem <= 0.05 then return nil end
        end
    end
    return nil
end

local function is_target_inside_building(ch, hrp)
    if not spawned or not (ch and hrp) then return false end
    local lch = lp.Character
    local lhrp = lch and (lch:FindFirstChild("HumanoidRootPart") or lch.PrimaryPart)
    if not lhrp then return false end
    local dist = (hrp.Position - lhrp.Position).Magnitude
    if dist > 150 or dist < 2 then return false end
    local ign = { lch, ch, ws:FindFirstChild("Entities") }
    if not ray_solid(lhrp.Position, hrp.Position - lhrp.Position, ign) then return false end
    if not ray_solid(hrp.Position, Vector3.new(0, 160, 0), ign) then return false end
    local hits = 0
    for _, d in ipairs(DIRS) do if ray_solid(hrp.Position, d * 85, ign) then hits += 1 end end
    return hits >= 2
end

local last_bldg_tp = 0
local active_mini_tp_orig = nil

local function try_teleport_inside_building(ch, hrp)
    if unloaded or not tp_bldg_on or not spawned or not (ch and hrp) or (os.clock() - last_bldg_tp) < 1.2 then return false end
    local intended_victim = (sel_plr and sel_plr ~= "None" and sel_plr) or cur_tgt
    if not intended_victim or ch.Name ~= intended_victim then return false end
    if is_spawn_protected(ch.Name) then return false end

    local lch = lp.Character
    local lhrp = lch and (lch:FindFirstChild("HumanoidRootPart") or lch.PrimaryPart)
    if not lhrp or not is_target_inside_building(ch, hrp) then return false end
    local ign = { lch, ch, ws:FindFirstChild("Entities") }
    for _, dist in ipairs({ 3.5, 5.0, 6.5 }) do
        for _, d in ipairs(DIRS) do
            local cand = hrp.Position + d * dist
            local fh = ray_solid(cand + Vector3.new(0, 4, 0), Vector3.new(0, -16, 0), ign)
            local pos = Vector3.new(cand.X, fh and (fh.Position.Y + 3) or hrp.Position.Y, cand.Z)
            bldg_ovp.FilterDescendantsInstances = ign
            local pts = ws:GetPartBoundsInBox(CFrame.new(pos + Vector3.new(0, 1, 0)), Vector3.new(2.2, 3.4, 2.2), bldg_ovp)
            local blocked = false
            for _, p in ipairs(pts) do if p.CanCollide then blocked = true; break end end
            if not blocked and ray_solid(pos + Vector3.new(0, 1.5, 0), Vector3.new(0, 160, 0), ign)
                and not ray_solid(pos + Vector3.new(0, 1.5, 0), (hrp.Position - pos), ign) then
                last_bldg_tp = os.clock()
                local cf = CFrame.lookAt(pos, Vector3.new(hrp.Position.X, pos.Y, hrp.Position.Z))
                lhrp.AssemblyLinearVelocity = Vector3.zero
                lhrp.CFrame = cf
                active_mini_tp_orig = cf
                add_log("Entered building -> " .. tostring(ch.Name))
                return true
            end
        end
    end
    return false
end

local function fire_normal(t_name, count)
    if unloaded or not spawned or not t_name or t_name == "None" then return false end
    if is_self_blocked() or is_spawn_protected(t_name) then return false end
    init_caches()
    if not (ability_hdl and ability_data and target_type) then return false end

    local h = rawget(ability_hdl, "activeAbility")
    local name = type(h) == "table" and rawget(h, "_name") or nil
    if type(name) ~= "string" or AIM_EXCLUDED[name] then return false end
    if is_ability_on_cooldown(name) or is_target_blocked(t_name, h, name, false) then return false end

    local ch, hrp = get_char(plrs:FindFirstChild(t_name))
    if not (ch and hrp) then return false end
    if not is_within_strict_spam_range(name, cur_eq_id, hrp) then return false end
    local d = ability_data[name]

    if tp_bldg_on then try_teleport_inside_building(ch, hrp) end
    local lch = lp.Character
    local my_root = lch and (lch:FindFirstChild("HumanoidRootPart") or lch.PrimaryPart)
    local my_real_pos = my_root and my_root.Position or nil

    local ts = rawget(h, "_targetSystem")
    local orig = ts and ts.Callback
    if ts then
        local k = ts.TargetType
        rawset(ts, "ValidTarget", ch)
        rawset(ts, "ValidTargetPoint", hrp.Position)
        rawset(ts, "ValidTargets", { [ch] = true })
        ts.Callback = function(_, v)
            if k == AREA_CHECK or k == CYLINDER_CAST then return v ~= nil and (v == ch and ch or nil) or { [ch] = true }
            elseif k == CAST_RESULT then return hrp.Position
            elseif v ~= nil then return v == ch and ch or nil end
            return ch, nil
        end
    end

    local cd_lbl = get_cd_label()
    local act_fn = h.activated
    local deact_fn = (d and d.holding) and h.deactivate or nil
    local n_fires = math.clamp(count or 1, 1, 25)
    local any_ok = false

    for _ = 1, n_fires do
        if unloaded or (cd_lbl and cd_lbl.Visible) then break end
        rawset(h, "_isHolding", true)
        if pcall(act_fn, h) then any_ok = true end
        if deact_fn then
            rawset(h, "_isHolding", false)
            pcall(deact_fn, h)
        end
    end

    if ts and orig then ts.Callback = orig end

    if any_ok and not (cd_lbl and cd_lbl.Visible) then
        play_ability_animation(name, cur_eq_id, my_real_pos, hrp)
    end
    return any_ok
end

local function fire_pingless_burst(t_name, count)
    if unloaded or not spawned or not t_name or t_name == "None" then return false end
    if is_self_blocked() or is_spawn_protected(t_name) then return false end
    init_caches()
    if not (ability_hdl and target_type and rem_activated and fire_server) then
        return fire_normal(t_name, count)
    end

    local h = rawget(ability_hdl, "activeAbility")
    local name = type(h) == "table" and rawget(h, "_name") or nil
    if type(name) ~= "string" or AIM_EXCLUDED[name] then return false end
    if is_ability_on_cooldown(name) or is_target_blocked(t_name, h, name, false) then return false end
    if force_normal[name] then return fire_normal(t_name, count) end

    local ch, hrp = get_char(plrs:FindFirstChild(t_name))
    if not (ch and hrp) then return false end
    if not is_within_strict_spam_range(name, cur_eq_id, hrp) then return false end

    local d = ability_data and ability_data[name]
    local ts = rawget(h, "_targetSystem")
    if not ts then return false end

    if tp_bldg_on then try_teleport_inside_building(ch, hrp) end
    local lch = lp.Character
    local my_root = lch and (lch:FindFirstChild("HumanoidRootPart") or lch.PrimaryPart)
    local my_real_pos = my_root and my_root.Position or nil

    local kind = rawget(ts, "TargetType") or ts.TargetType
    local timeout = is_motus_equipped(name) and 15 or (rawget(h, "_timeout") or h._timeout or (d and (d.cooldown or d.timeout or d.Cooldown)) or 1)

    rawset(ts, "ValidTarget", ch)
    rawset(ts, "ValidTargetPoint", hrp.Position)
    rawset(ts, "ValidTargets", { [ch] = true })

    local orig_cb = ts.Callback
    ts.Callback = function(_, v)
        if kind == AREA_CHECK or kind == CYLINDER_CAST then return v ~= nil and (v == ch and ch or nil) or { [ch] = true }
        elseif kind == CAST_RESULT then return hrp.Position
        elseif v ~= nil then return v == ch and ch or nil end
        return ch, nil
    end
    local ok, info = pcall(ts.GetTargetInfo, ts)
    ts.Callback = orig_cb

    if not ok or info == nil or (typeof(info) == "table" and next(info) == nil) then
        if kind == AREA_CHECK or kind == CYLINDER_CAST then info = { [ch] = true }
        elseif kind == CAST_RESULT then info = hrp.Position
        else info = ch end
    end

    local it = typeof(info)
    if not (it == "Vector3" or it == "Instance" or (it == "table" and next(info) ~= nil)) then
            return false
    end

    local cd_lbl = get_cd_label()
    local n_fires = math.clamp(count or 1, 1, 35)
    local any_sent = false
    if it == "Vector3" then
        for _ = 1, n_fires do
            if unloaded or (cd_lbl and cd_lbl.Visible) then break end
            if pcall(fire_server, rem_activated, info, generate_guid(http, false)) then
                any_sent = true
            end
        end
    else
        for _ = 1, n_fires do
            if unloaded or (cd_lbl and cd_lbl.Visible) then break end
            if pcall(fire_server, rem_activated, info) then
                any_sent = true
            end
        end
    end

    if not any_sent then return false end

    if is_in_animation_range(name, cur_eq_id, my_real_pos, hrp) then
        local was_alive = client_debounce and type(client_debounce.isAlive) == "function" and client_debounce.isAlive(name)
        if not was_alive and not is_ability_on_cooldown(name) then
            play_ability_animation(name, cur_eq_id, my_real_pos, hrp)
            if type(timeout) == "number" and timeout > 0 and client_debounce and type(client_debounce.set) == "function" then
                pcall(client_debounce.set, name, timeout)
            end
            if bind_activated and fire_bind then
                pcall(fire_bind, bind_activated, name)
            end
        end
    end

    return true
end

local function cast_proj(TARGET_NAME, burst_count)
    if unloaded or proj_casting or not spawned or not TARGET_NAME or TARGET_NAME == "None" then return false end
    if is_self_blocked() or is_spawn_protected(TARGET_NAME) then return false end

    init_caches()
    if not (ability_hdl and ability_data and target_type) then return false end

    local h = rawget(ability_hdl, "activeAbility")
    if type(h) ~= "table" then return false end
    local ab_name = rawget(h, "_name")
    if type(ab_name) ~= "string" or AIM_EXCLUDED[ab_name] then return false end

    if is_ability_on_cooldown(ab_name) or is_target_blocked(TARGET_NAME, h, ab_name, true) then
        return false
    end

    local target_plr = plrs:FindFirstChild(TARGET_NAME)
    local char, hrp = get_char(target_plr)
    if not char then char = target_plr and target_plr.Character end
    local a = char and anchor(char)
    if not (char and a) then return false end

    if tp_bldg_on and hrp then
        try_teleport_inside_building(char, hrp)
    end

    proj_casting = true
    cur_tgt_ch = char
    cur_tgt_hrp = hrp or a
    write_target_on(h)

    local ts = rawget(h, "_targetSystem")
    local orig_cb = ts and ts.Callback
    if type(ts) == "table" then
        local kind = rawget(ts, "TargetType")
        rawset(ts, "ValidTarget", char)
        rawset(ts, "ValidTargetPoint", a.Position)
        rawset(ts, "ValidTargets", { [char] = true })
        ts.Callback = function(_, victim)
            if kind == AREA_CHECK or kind == CYLINDER_CAST then
                return { [char] = true }
            elseif kind == CAST_RESULT then
                return a.Position
            elseif victim ~= nil then
                return victim == char and char or nil
            end
            return char, nil
        end
    end

    local d = ability_data[ab_name]
    local n_fires = math.clamp(burst_count or 1, 1, 15)
    local any_ok = false
    for _ = 1, n_fires do
        if unloaded or is_ability_on_cooldown(ab_name) or is_target_blocked(TARGET_NAME, h, ab_name, true) then break end
        rawset(h, "_isHolding", true)
        if pcall(function() h:activated() end) then any_ok = true end
        if d and d.holding then
            task.wait()
            rawset(h, "_isHolding", false)
            pcall(function() h:deactivate() end)
        end
    end

    if ts and orig_cb then ts.Callback = orig_cb end
    proj_casting = false
    return any_ok
end

trigger_instant_spam = function()
    if unloaded or not spawned or not int_on or not (spam_on or priv_spam_on) or cps <= 0 then return end
    if is_self_blocked() then return end
    local t = (sel_plr ~= "None" and sel_plr) or cur_tgt
    if not t or t == "None" or is_spawn_protected(t) then return end
    local h = ability_hdl and rawget(ability_hdl, "activeAbility")
    local nm = type(h) == "table" and rawget(h, "_name") or nil
    if is_ability_on_cooldown(nm) or is_target_blocked(t, h, nm, proj_on) then return end
    local open_burst = math.clamp(math.ceil(cps / 60), 1, 35)
    if proj_on then
        if not ign_proj then
            task.spawn(cast_proj, t, math.clamp(math.ceil(cps / 120), 1, 8))
        end
    elseif spam_on then
        pcall(fire_pingless_burst, t, open_burst)
    elseif priv_spam_on then
        pcall(fire_normal, t, open_burst)
    end
end

local function find_hover_target()
    if sel_plr and sel_plr ~= "None" then
        local p = plrs:FindFirstChild(sel_plr)
        local ch, hrp = get_char(p)
        if ch and hrp then return p.Name, ch, hrp, check_vis(ch, hrp) end
        return nil, nil, nil, false
    end
    cam = ws.CurrentCamera
    if not cam then return nil, nil, nil, false end
    local mp = uis:GetMouseLocation()
    local vr = cam:ViewportPointToRay(mp.X, mp.Y)
    local ro, rd = vr.Origin, vr.Direction.Unit

    if resp_barriers then
        m_rp.FilterDescendantsInstances = lp.Character and { lp.Character } or {}
        local hit = ws:Raycast(ro, rd * 2500, m_rp)
        if hit and hit.Instance then
            for _, p in ipairs(plrs:GetPlayers()) do
                local ch, hrp = get_char(p)
                if ch and hrp and hit.Instance:IsDescendantOf(ch) and check_vis(ch, hrp) then
                    return p.Name, ch, hrp, true
                end
            end
        end
    end

    local bn, bc, bh, bd = nil, nil, nil, 4.5
    for _, p in ipairs(plrs:GetPlayers()) do
        local ch, hrp = get_char(p)
        if ch and hrp then
            local proj = (hrp.Position - ro):Dot(rd)
            if proj > 0 and proj <= 1000 then
                local d = (hrp.Position - (ro + rd * proj)).Magnitude
                if d < bd and check_vis(ch, hrp) then
                    bd, bn, bc, bh = d, p.Name, ch, hrp
                end
            end
        end
    end
    return bn, bc, bh, bn ~= nil
end

mob.find_center = function()
    cam = ws.CurrentCamera
    if not cam then return nil, nil, nil, false end
    local vp = cam.ViewportSize
    local center = Vector2.new(vp.X * 0.5, vp.Y * 0.5)
    local vr = cam:ViewportPointToRay(center.X, center.Y)
    local ro, rd = vr.Origin, vr.Direction.Unit

    if resp_barriers then
        m_rp.FilterDescendantsInstances = lp.Character and { lp.Character } or {}
        local hit = ws:Raycast(ro, rd * 2500, m_rp)
        if hit and hit.Instance then
            for _, p in ipairs(plrs:GetPlayers()) do
                local ch, hrp = get_char(p)
                if ch and hrp and hit.Instance:IsDescendantOf(ch) and check_vis(ch, hrp) then
                    return p.Name, ch, hrp, true
                end
            end
        end
    end

    local bn, bc, bh, bd = nil, nil, nil, 14
    for _, p in ipairs(plrs:GetPlayers()) do
        local ch, hrp = get_char(p)
        if ch and hrp and check_vis(ch, hrp) then
            local proj = (hrp.Position - ro):Dot(rd)
            if proj > 0 and proj <= 1000 then
                local d = (hrp.Position - (ro + rd * proj)).Magnitude
                if d < bd then
                    bd, bn, bc, bh = d, p.Name, ch, hrp
                end
            end
        end
    end
    return bn, bc, bh, bn ~= nil
end

local function find_target()
    if mob.on then
        if sel_plr and sel_plr ~= "None" then
            local p = plrs:FindFirstChild(sel_plr)
            local ch, hrp = get_char(p)
            if ch and hrp then return p.Name, ch, hrp, check_vis(ch, hrp) end
            return nil, nil, nil, false
        end
        if mob.lock then
            if mob.tgt then
                local p = plrs:FindFirstChild(mob.tgt)
                local ch, hrp = get_char(p)
                if ch and hrp then
                    mob.ch, mob.hrp = ch, hrp
                    return mob.tgt, ch, hrp, check_vis(ch, hrp)
                else
                    mob.tgt, mob.ch, mob.hrp = nil, nil, nil
                end
            end
            local bn, bc, bh, ok_l = mob.find_center()
            if bn and bc and bh then
                mob.tgt, mob.ch, mob.hrp = bn, bc, bh
            end
            return bn, bc, bh, ok_l
        end
        return nil, nil, nil, false
    end
    if sel_plr and sel_plr ~= "None" then return find_hover_target() end
    cam = ws.CurrentCamera
    if not cam then return nil, nil, nil, false end
    local mp = uis:GetMouseLocation()
    local vr = cam:ViewportPointToRay(mp.X, mp.Y)
    local ro, rd = vr.Origin, vr.Direction.Unit
    local lhrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    local spos = lhrp and lhrp.Position or ro

    m_rp.FilterDescendantsInstances = lp.Character and { lp.Character } or {}
    local hit = resp_barriers and ws:Raycast(ro, rd * 2500, m_rp) or nil
    local hp = hit and hit.Position or nil

    local bn, bc, bh, bs = nil, nil, nil, math.huge
    for _, p in ipairs(plrs:GetPlayers()) do
        local ch, hrp = get_char(p)
        if ch and hrp and check_vis(ch, hrp) then
            local pos = hrp.Position
            local ds = (pos - spos).Magnitude
            local proj = (pos - ro):Dot(rd)
            local dr = (pos - (proj > 0 and (ro + rd * proj) or ro)).Magnitude
            local rs_d = math.min(dr, hp and (pos - hp).Magnitude or dr)
            if ds <= 115 or rs_d <= 100 then
                local sp, on_s = cam:WorldToViewportPoint(pos)
                local sd = (Vector2.new(sp.X, sp.Y) - mp).Magnitude
                local scr = (sd * 0.08) + (rs_d * 0.85) + (ds * 0.35) + (on_s and 0 or 150)
                if cur_tgt == p.Name then scr *= 0.9 end
                if scr < bs then bs, bn, bc, bh = scr, p.Name, ch, hrp end
            end
        end
    end
    return bn, bc, bh, bn ~= nil
end

local function over_ui_or_prompt()
    if uis:GetFocusedTextBox() then return true end
    local mp = uis:GetMouseLocation()
    local inset = gui_svc:GetGuiInset()
    local gx, gy = mp.X - inset.X, mp.Y - inset.Y

    local pg = lp:FindFirstChild("PlayerGui")
    if pg then
        local ok, objs = pcall(pg.GetGuiObjectsAtPosition, pg, gx, gy)
        if ok and type(objs) == "table" then
            for _, obj in ipairs(objs) do
                if obj.Visible and (obj:IsA("GuiButton") or obj:IsA("TextBox")) then
                    return true
                end
            end
        end
    end

    local hp = gui_parent()
    if hp and hp ~= pg and hp:IsA("BasePlayerGui") then
        local ok, objs = pcall(hp.GetGuiObjectsAtPosition, hp, gx, gy)
        if ok and type(objs) == "table" then
            for _, obj in ipairs(objs) do
                if obj.Visible and (obj:IsA("GuiButton") or obj:IsA("TextBox")) then
                    return true
                end
            end
        end
    end

    local raw_t = MOUSE_HOLDER.r.Target
    if raw_t and (raw_t:FindFirstChildOfClass("ClickDetector") or raw_t:FindFirstChildOfClass("ProximityPrompt")
        or (raw_t.Parent and (raw_t.Parent:FindFirstChildOfClass("ClickDetector") or raw_t.Parent:FindFirstChildOfClass("ProximityPrompt")))) then
        return true
    end
    return false
end

local exclusive_bound = false

        exclusive_bound = false

local function handle_mouse_input(_, inputState, _)
    if unloaded then
        pcall(function() cas:UnbindAction(ACTION_NAME) end)
        return Enum.ContextActionResult.Pass
    end
    if over_ui_or_prompt() then
        return Enum.ContextActionResult.Pass
    end

    if inputState == Enum.UserInputState.Begin then
        local t = (sel_plr ~= "None" and sel_plr) or cur_tgt
        if t and t ~= "None" and not is_spawn_protected(t) and spawned and hb_on and proj_on and ign_proj then
            task.spawn(function()
                cast_proj(t, 1)
            end)
        end
        return Enum.ContextActionResult.Sink
    end

    return Enum.ContextActionResult.Sink
end

local function bind_mouse_lock()
    if not unloaded and not exclusive_bound then
        exclusive_bound = true
        cas:BindActionAtPriority(
            ACTION_NAME,
            handle_mouse_input,
            false,
            Enum.ContextActionPriority.High.Value + 100,
            Enum.UserInputType.MouseButton1
        )
    end
end

    local function overlapping(a, b)
        if not (a and b and a.AbsolutePosition and b.AbsolutePosition) then return false end
        local ax, ay = a.AbsolutePosition.X, a.AbsolutePosition.Y
        local aw, ah = a.AbsoluteSize.X, a.AbsoluteSize.Y
        local bx, by = b.AbsolutePosition.X, b.AbsolutePosition.Y
        local bw, bh = b.AbsoluteSize.X, b.AbsoluteSize.Y
        return ax < bx + bw and bx < ax + aw and ay < by + bh and by < ay + ah
    end

    local function legit_complete_prompt(inst)
        task.spawn(function()
            local safety = 0
            local is_sc = type(inst.cursorRef) == "table" and type(inst.targetRef) == "table"
                and type(inst.cursorRef.get) == "function" and type(inst.targetRef.get) == "function"
            local last_hit = 0
            while not unloaded and safety < 2000 and alive_prompt(inst) do
                safety = safety + 1
                if is_sc then
                    local ok1, cursor = pcall(function() return inst.cursorRef:get() end)
                    local ok2, target = pcall(function() return inst.targetRef:get() end)
                    if ok1 and ok2 and cursor and target and overlapping(cursor, target)
                        and (os.clock() - last_hit) > 0.18 then
                        pcall(function() inst:onInputBegan() end)
                        last_hit = os.clock()
                    end
                    task.wait()
                else
                    if type(inst.onInputBegan) == "function" then
                        pcall(function() inst:onInputBegan() end)
                    elseif inst.inputBegan then
                        local step = (type(inst.progressIncrease) == "number" and inst.progressIncrease) or 0.5
                        local p = math.clamp((inst.progress or 0) + step, 0, 5)
                        inst.progress = p
                        if inst.progressValue then pcall(function() inst.progressValue:set(p) end) end
                        local ok, done = pcall(inst.inputBegan, p, false)
                        if ok and done == true then inst.hasEscaped = true; return end
                    else return end
                    task.wait(0.09 + math.random() * 0.1)
                end
            end
        end)
    end

    local function sweep_prompts()
        local pending = {}
        for _, t in ipairs(targets) do if not t.done then pending[t.tail] = t end end
        if not next(pending) then return true end
        local by_tail = {}
        if type(filtergc) == "function" then
            local ns = filtergc("function", { Name = "new", IgnoreExecutor = true }) or {}
            for _, fn in ipairs(ns) do
                local ok_s, s = pcall(debug.info, fn, "s")
                if ok_s and type(s) == "string" then
                    for tail in pairs(pending) do
                        if not by_tail[tail] and s:sub(-#tail) == tail then by_tail[tail] = fn end
                    end
                end
            end

            for tail, fn in pairs(by_tail) do
                local tables = filtergc("table", { KeyValuePairs = { new = fn } }) or {}
                if tables[1] then cls[tail] = tables[1] end
            end
            for tail, class in pairs(cls) do
                local t = pending[tail]; if t and not t.done then
                    local real = class.new
                    local key, completer = t.key, t.fn
                    class.new = function(...)
                        local inst = real(...)
                        if type(inst) ~= "table" then return inst end
                        if auto[key] then task.defer(completer, inst) end
                        return inst
                    end
                    auto.restore[#auto.restore + 1] = function() class.new = real end
                    t.done = true
                end
            end
        elseif type(getgc) == "function" then
            for _, tbl in ipairs(getgc(true)) do
                if type(tbl) == "table" and type(rawget(tbl, "new")) == "function" then
                    local ok_s, s = pcall(debug.info, rawget(tbl, "new"), "s")

                                end
                                auto.restore[#auto.restore + 1] = function() tbl.new = real end
                                t.done = true
                            end
                        end
                    end
                end
            end
        end
        for _, t in ipairs(targets) do if not t.done then return false end end

win._was_blocked = true
win._prev_list = {}

win._step_outspammer = function(t_name)
    if unloaded or not (spam_on or priv_spam_on) or not int_on or cps <= 0 then return end
    if not t_name or t_name == "None" or is_self_blocked() or is_spawn_protected(t_name) then
        win._was_blocked = true
        return
    end

    local h = ability_hdl and rawget(ability_hdl, "activeAbility")
    local nm = type(h) == "table" and rawget(h, "_name") or nil
    if is_ability_on_cooldown(nm) or is_target_blocked(t_name, h, nm, proj_on) then
        win._was_blocked = true
        return
    end

    local ch, hrp = get_char(plrs:FindFirstChild(t_name))
    if not (ch and hrp) or not is_within_strict_spam_range(nm, cur_eq_id, hrp) then
        return
    end

    local now = os.clock()
    local dt = (osp_last_step > 0) and math.clamp(now - osp_last_step, 0, 0.05) or (1 / 180)
    osp_last_step = now

    if win._was_blocked then
        win._was_blocked = false
        osp_acc = math.max(1, math.ceil(cps / 120))
    else
        osp_acc = osp_acc + (dt * cps)
    end

    local fires = math.floor(osp_acc)
    if fires <= 0 then return end
    osp_acc = osp_acc - fires

    if proj_on then
        if not ign_proj and not proj_casting then
            task.spawn(cast_proj, t_name, math.clamp(fires, 1, 12))
        end
    elseif spam_on then
        pcall(fire_pingless_burst, t_name, fires)
    elseif priv_spam_on then
        pcall(fire_normal, t_name, fires)
    end
end

win._fast_step = function()
    if unloaded or not spawned or not int_on or not (spam_on or priv_spam_on) then return end
    if sel_plr and sel_plr ~= "None" then
        local p = plrs:FindFirstChild(sel_plr)
        local ch, hrp = get_char(p)
        if ch and hrp and check_vis(ch, hrp) then
            cur_tgt, cur_tgt_ch, cur_tgt_hrp = sel_plr, ch, hrp
            win._step_outspammer(sel_plr)
        end
    elseif cur_tgt then
        win._step_outspammer(cur_tgt)
    end
end

add_conn(run.RenderStepped:Connect(function()
    if unloaded then return end
    if mob.on and spawned and hb_on and mob.lock and int_on then
        local lock_name = (sel_plr and sel_plr ~= "None" and sel_plr) or mob.tgt
        if not lock_name or lock_name == "None" then
            local bn, bc, bh = mob.find_center()
            if bn and bc and bh then
                mob.tgt, mob.ch, mob.hrp = bn, bc, bh
                lock_name = bn
            end
        end
        if lock_name and lock_name ~= "None" then
            local p = plrs:FindFirstChild(lock_name)
            local ch, hrp = get_char(p)
            if ch and hrp then
                cam = ws.CurrentCamera
                if cam then
                    local cam_pos = cam.CFrame.Position
                    local tgt_pos = hrp.Position
                    if (tgt_pos - cam_pos).Magnitude > 0.1 then
                        cam.CFrame = CFrame.lookAt(cam_pos, tgt_pos)
                    end
                end
            else
                if lock_name == mob.tgt then
                    mob.tgt, mob.ch, mob.hrp = nil, nil, nil
                end
            end
        end
    end
    win._fast_step()
end))

add_conn(run.Heartbeat:Connect(function()
    if unloaded then return end
    spawned = is_spawned()
    if not spawned then
        int_on, proj_on = false, false
        clr_all()
        if dbg_ui and stat_lbl then
            stat_lbl.Text = "Spawned: NO | Ability Equipped: NO\nTarget: NONE | Mode: PAUSED"
        end
        return
    end

    update_magic_stat_watcher()

    if is_motus_equipped() and is_ability_on_cooldown("Motus") then
        stopMotusAnimation()
    end

    local v_list, s_map = { "None" }, {}
    for _, p in ipairs(plrs:GetPlayers()) do
        local ch, hrp = get_char(p)
        if ch and hrp then table.insert(v_list, p.Name); s_map[p.Name] = true end
    end
    if sel_plr and sel_plr ~= "None" and not s_map[sel_plr] and not is_spawn_protected(sel_plr) then
        local lost = sel_plr
        win._dd_int = true; sel_plr = "None"
        pcall(function() mob.lock_dd:Set({"None"}) end)
        pcall(function() mob.lock_dd:Set("None") end)
        win._dd_int = false
        local msg = lost .. (plrs:FindFirstChild(lost) and " has died, target lock reset." or " has left, target lock reset.")
        win:Notify({ title = "Target Lock", Title = "Target Lock", content = msg, Content = msg, duration = 6, Duration = 6 })
    end

    local chg = #v_list ~= #win._prev_list
    if not chg then for i, n in ipairs(v_list) do if n ~= win._prev_list[i] then chg = true; break end end end
    if chg then mob.lock_dd:Refresh(v_list); win._prev_list = v_list end

    local ok_ab, is_pr, eq_id, raw_img = get_ability()
    int_on, proj_on = ok_ab, is_pr

    local h = ability_hdl and rawget(ability_hdl, "activeAbility")
    local nm = type(h) == "table" and rawget(h, "_name") or nil
    local has_no_hl_aoe = is_no_highlight_aoe_equipped(nm)

    if not int_on then
        unbind_mouse_lock()
    end

    local any_spam = spam_on or priv_spam_on
    if not (hb_on or any_spam) or not int_on then
        clr_all()
        if not has_no_hl_aoe then
            remove_non_target_highlights(nil)
        end
        return
    end

    local t_name, t_ch, t_hrp, ok_los
    if hb_on then
        t_name, t_ch, t_hrp, ok_los = find_target()
    else
        t_name, t_ch, t_hrp, ok_los = find_hover_target()
    end
    cur_tgt, cur_tgt_ch, cur_tgt_hrp = t_name, t_ch, t_hrp

    local target_for_aim = (sel_plr ~= "None" and sel_plr) or (ok_los and cur_tgt or nil)

    if type(h) == "table" and t_ch and t_hrp then
        write_target_on(h)
    end

    if proj_on then
        sync_proj_assist(target_for_aim)
        if any_spam and not ign_proj and target_for_aim and not proj_casting and not is_spawn_protected(target_for_aim) then
            win._step_outspammer(target_for_aim)
        end
    else
        unbind_mouse_lock()
        win._step_outspammer(target_for_aim)
    end

    if hb_on and t_name and t_ch and t_hrp and ok_los then
        if has_no_hl_aoe then
            clr_hl()
        else
            set_hl(t_ch)
            remove_non_target_highlights(t_ch)
        end
        if vis_on then
            if not mob.bbg.Parent then mob.bbg.Parent = gui_parent() end
            mob.bbg.Adornee = t_hrp
            mob.bbg.Enabled = true
        else
            mob.bbg.Enabled = false; mob.bbg.Adornee = nil; mob.out_c.Visible = false
        end
    else
        mob.bbg.Enabled = false; mob.bbg.Adornee = nil; mob.out_c.Visible = false; clr_hl()
        if not has_no_hl_aoe then
            remove_non_target_highlights(nil)
        end
    end

    if dbg_ui and stat_lbl then
        local cd_s = is_ability_on_cooldown(nm) and "CD: YES" or "CD: NO"
        local blk_s = (is_self_blocked() or (t_name and is_target_blocked(t_name, h, nm, proj_on))) and "BLOCKED" or "READY"
        stat_lbl.Text = string.format(
            "Spawned: YES | Equipped: %s (%s)\nImage: %s | Proj: %s\nTarget: %s (STATE: %s) | LOS: %s",
            eq_id and "YES" or "NO", cd_s, tostring(raw_img or "NONE"),
            proj_on and "YES" or "NO", t_name or "NONE", blk_s, (t_name and ok_los) and "CLEAR" or "BLOCKED"
        )
    end
end))
